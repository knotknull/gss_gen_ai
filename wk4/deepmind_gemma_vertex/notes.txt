## Ray hello world 

import ray

ray.ini()       # initialize a ray session on ray cluster

@ray.remote     # convert function to a task
def hello_world(): 
    return "hello world"

future = hello_world.remote()   # execute the function on cluster, 
                                # returns a future (eventual result of asynchronous operation)
data = ray.get(future)          # return the result
print(data)
